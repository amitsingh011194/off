import pytest
import os
import datetime

def fetch_tags(tags_to_run):
    
    modified_list = []
    for tag in tags_to_run[:-1]:
        modified_list.append(tag + " or")
    modified_list.append(tags_to_run[-1])
    final_tags_to_run = " ".join(modified_list)
    return final_tags_to_run

def run_tests():

    timestamp = datetime.datetime.now().strftime("%Y-%m-%d_%H-%M-%S")
    
    reports_dir = "reports/log_reports"
    reports_html_dir = "reports/html_reports"
    
    if not os.path.exists(reports_dir):
        os.makedirs(reports_dir)
    elif not os.path.exists(reports_html_dir):
        os.makedirs(reports_html_dir)

    pytest_report_path = os.path.join(reports_dir, f"log_report_{timestamp}.html")
    pytest_html_reporter_path = os.path.join(reports_html_dir, f"pytest_html_report_{timestamp}.html")
    
    # tags_to_run = [
        
    #     'FNBO_ACH_Scheduled_RegularInstallment or FNBO_ACH_Scheduled_LessThanRegularInstallment',
        # 'FNBO_ACH_Scheduled_LessThanRegularInstallment',
        
    # ]
    
    # fnbo_tags_to_run = ["""
                        
    #     FNBO_ACH_Scheduled_RegularInstallment or
    #     FNBO_ACH_Scheduled_LessThanRegularInstallment
        
    # """]

    # single_line_string = fnbo_tags_to_run[0].replace("\n", " ")
    
    fnbo_tags = [
        
        'FNBO_ACH_Scheduled_RegularInstallment',
        'FNBO_ACH_Scheduled_LessThanRegularInstallment',
        # 'FNBO_ACH_Scheduled_PastDueAmount',
        # 'FNBO_ACH_Scheduled_LessThanPastDueAmount',
        
    ]
    
    tags_to_run = fetch_tags(fnbo_tags)
    
    pytest.main([
        '-k',
        tags_to_run,
        '--maxfail=15',
        '--disable-warnings',
        f'--html={pytest_report_path}',        
        f'--html-report={pytest_html_reporter_path}'
    ])

if __name__ == "__main__":
    run_tests()
