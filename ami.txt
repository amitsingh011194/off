def switch_to_tab_with_url(driver, partial_url):
    for handle in driver.window_handles:
        driver.switch_to.window(handle)
        if partial_url in driver.current_url:
            return True
    return False

@when('the user uploads 7 PGP-encrypted files to "<inbound_bucket>"')
def user_uploads_encrypted_files(config):
    current_date = datetime.now().strftime('%Y%m%d')
    local_dir = r"C:\Users\aitha253418\PayMentor_Docker\encryption"
    bucket = config['inbound_bucket']
    profile = 'qa'
    file_names = [
        f"CoBorrowerData_{current_date}_090735.csv.gpg",
        f"CustomerData_{current_date}_085342.csv.gpg",
        f"LoanInformation_{current_date}_083426.csv.gpg",
        f"LoanProfile_{current_date}_082748.csv.gpg",
        f"sfcEXLReconciliation_{current_date}_090028.csv.gpg",
        f"SSPLogin_{current_date}_084727.csv.gpg",
        f"PaymentData_{current_date}_084059.csv.gpg"
    ]

    if bucket.endswith('/'):
        bucket = bucket[:-1]

    for file_name in file_names:
        local_path = os.path.join(local_dir, file_name)
        if not os.path.exists(local_path):
            raise FileNotFoundError(f"File {local_path} not found")
        s3_key = f"{bucket}/{file_name}"
        cmd = [
            "aws", "s3", "cp",
            local_path,
            s3_key,
            "--profile", profile,
            "--no-verify-ssl"
        ]
        print(f"Uploading {local_path} to {s3_key}")
        result = subprocess.run(cmd, capture_output=True, text=True)
        if result.returncode != 0:
            print("STDOUT:", result.stdout)
            print("STDERR:", result.stderr)
            raise subprocess.CalledProcessError(result.returncode, cmd)
        # Screenshot not applicable here unless you have a browser context

        time.sleep(10)

# ---- CloudWatch Lambda Logs Steps ----

LOGS_FOLDER = r"C:\Users\aitha253418\PayMentor_Docker\logs"
PROFILE = 'qa'
REGION = 'us-east-1'

def open_cloudwatch_log_group(driver, url):
    driver.execute_script(f"window.open('{url}', '_blank');")
    take_step_screenshot(driver, "open_cloudwatch_log_group")
    time.sleep(2)
    driver.switch_to.window(driver.window_handles[-1])
    try:
        WebDriverWait(driver, 30).until(
            EC.presence_of_element_located((By.CSS_SELECTOR, "input[placeholder='Search log streams']"))
        )
        print(f"CloudWatch UI loaded: {url}")
    except Exception as e:
        print(f"CloudWatch UI did not load as expected: {e}")
    print(f"Opened CloudWatch log group UI: {url}")

def start_log_save_thread(log_group):
    thread = Thread(
        target=save_latest_lambda_log_to_file,
        args=(log_group, PROFILE, REGION, LOGS_FOLDER)
    )
    thread.start()
    return thread

@then('the user checks the specific tenant gatekeeper lambda logs using CloudWatch')
def check_gatekeeper_lambda_logs(main_page):
    log_group = "/aws/lambda/sb-utp1-1674e330-etl_gatekeeper"
