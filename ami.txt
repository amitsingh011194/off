import { NextResponse } from 'next/server';
import { HttpStatusCode } from 'axios';

import { EXTERNAL_API_ROUTES, API_MESSAGES } from '@/lib/constants';
import { headers } from 'next/headers';

export async function GET() {
  const url = `${EXTERNAL_API_ROUTES.FEATURE_FLAG.base}`;
  const headersList: any = headers();
  const requestOptions: RequestInit = {
    method: 'GET',
    headers: { ...headersList },
    cache: 'no-store',
    redirect: 'follow',
  };

  const res = await fetch(url, requestOptions).catch((error: any) => {
    const data = {
      message: error.message || API_MESSAGES.SOMETHING_WENT_WRONG,
      statusCode: error.status || HttpStatusCode.InternalServerError,
      error,
    };
    return NextResponse.json(data, { status: data.statusCode });
  });
  const response = await res.json();

  return NextResponse.json(response, { status: response.statusCode });
}
