@given('AWS credentials are refreshed for profile "qa"')
def aws_credentials_refreshed():
    profile = "qa"

    # Check if the profile exists
    result = subprocess.run(["aws", "configure", "list-profiles"], capture_output=True, text=True)
    profiles = result.stdout.splitlines()

    if profile not in profiles:
        pytest.skip(f"AWS profile '{profile}' not found. Skipping test.")

    # Attempt SSO login
    try:
        subprocess.run(["aws", "sso", "login", "--profile", profile], check=True)
    except subprocess.CalledProcessError as e:
        pytest.skip(f"SSO login failed for profile '{profile}': {e}")
