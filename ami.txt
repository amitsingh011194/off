#!/bin/bash


# Check if three command-line arguments are provided
if [ "$#" -ne 1 ]; then
 echo "-- Error: Provide lambda version to build"
 echo "-- Usage: $0 "
 exit 1
fi

# Zip File naming
VERSION=$1
echo $VERSION

APP_DIR="$(pwd)/bu-dgt-paymentor-core-aws-app/application"

# Define the S3 bucket and prefix
S3_BUCKET="${AWS_ACCOUNT_ID}-paymentor-tf-state-mgmt"
S3_PREFIX="artifacts/${TENANT_ID}/${VERSION}"

#
# Zip Lambdas
#

build_java_lambda(){
 cd ${APP_DIR}/lambdas/CommunicationIVREventsSaveAudio/

 echo "Running Maven build inside Docker..."

 DOCKER_IMAGE="maven:3.9.8-amazoncorretto-11"
 WORKSPACE_DIR="$(pwd)"
 MAVEN_COMMAND="mvn clean package"

 docker run --rm \
 -u $(id -u):$(id -g) \
 -v "${WORKSPACE_DIR}:/usr/src/app" \
 -w /usr/src/app \
 "${DOCKER_IMAGE}" \
 bash -c "${MAVEN_COMMAND}"

 echo "Maven build completed successfully."

 ls -l *

 cp target/kv-audio-lambda-project-1.0-SNAPSHOT.jar ${WORK_DIR}/lambdas
}

build_lambdas(){
 cd ${APP_DIR}/lambdas

 echo "Zipping Lambdas"
 for lambda_name in `ls`; do
 zip -r "${WORK_DIR}/lambdas/$lambda_name-${VERSION}.zip" "$lambda_name"
 done
 echo "Zipping complete!"
}

build_lex_bots(){
 cd ${APP_DIR}/lex

 cp -r * ${WORK_DIR}/lex
}

build_flyway_scripts(){
 cd ${APP_DIR}/db

 cp *.sql ${WORK_DIR}/flyway
}

cleanup(){
 echo "--- Deleting ${WORK_DIR}"
 rm -rf ${WORK_DIR}
}

main(){
 # Setup the tmp dir
 WORK_DIR=$(mktemp -d)
 trap cleanup EXIT # Delete temp dir on the way out

 cd ${WORK_DIR}
 mkdir lambdas lex flyway

 if ${BUILD_JAVA_LAMBDA}; then
 build_java_lambda
 fi

 build_lambdas
 build_lex_bots
 build_flyway_scripts
 
 ls -l ${WORK_DIR}/*

 #
 # Ship to S3
 #
 echo "Shipping zips to S3"
 aws s3 sync ${WORK_DIR} "s3://${S3_BUCKET}/${S3_PREFIX}/"
 echo "Shipping complete!"
}
main
